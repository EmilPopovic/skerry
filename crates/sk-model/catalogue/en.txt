# The English catalogue. One `key = value` per line; `#` starts a comment.
#
# Every user-visible string in Skerry resolves through here from the first
# commit — not because localization is an early priority, but because
# retrofitting string extraction across twelve crates costs weeks, and because
# the discipline catches sentences assembled by concatenation, which cannot be
# translated.
#
# `{0}`, `{1}` are positional and may be reordered by a translation.

app.name = Skerry

# Column headers, Details view.

column.size = Size

# The status line. Numbers are positional so a translation can reorder them.

status.empty = This folder is empty
status.filtered = {0} of {1} items match “{2}”

# Sizes. A file manager writes these thousands of times per session, so the
# unit is a separate string rather than concatenated onto a number.

# Errors reaching a human say what happened and what to do, and are separate
# from the debug chain that goes to the log.

error.no_display = Cannot reach a display server. Set WAYLAND_DISPLAY or DISPLAY and try again.
error.no_gpu = No usable GPU. Install a Vulkan driver, or mesa's software renderer as a fallback.
error.no_font = No usable font found. Install a font package such as dejavu-fonts or noto-fonts.
